    "detected_low": Level.NOTE,
    "suspicious": Level.WARNING,

def _build_rules_from_patterns(pattern_engine: Any) -> list[ReportingDescriptor]:
    """Convert all loaded pattern rules to SARIF reportingDescriptors."""
    # Enumerate pattern_engine rules to create descriptors
    ...

def _compute_fingerprint(rule_id: str, file_path: str, matched_text: str) -> str:
    """SHA-256 fingerprint for deduplication across runs."""
    return hashlib.sha256(f"{rule_id}:{file_path}:{matched_text}".encode()).hexdigest()

def build_sarif(
    scan_results: list[dict[str, Any]],
    rules: list[ReportingDescriptor],
) -> Sarif:
    """Build a SARIF 2.1.0 document from scan results (D-08, D-09)."""
    results: list[Result] = []
    for sr in scan_results[:_MAX_RESULTS]:
        verdict = sr.get("verdict", "clean")
        if verdict == "clean":
            continue
        severity = sr.get("severity", "")
        level_key = f"{verdict}_{severity}" if severity else verdict
        level = _VERDICT_SEVERITY_TO_LEVEL.get(level_key, Level.WARNING)

        results.append(Result(
            rule_id=sr.get("rule_id", "SEMANTIC"),
            level=level,
            message=Message(text=sr.get("message", "Detection finding")),
            locations=[Location(
                physical_location=PhysicalLocation(
                    artifact_location=ArtifactLocation(uri=sr.get("file_path", "")),
                    region=Region(start_line=sr.get("line_number", 1)),
                )
            )],
            partial_fingerprints={
                "primaryLocationLineHash": _compute_fingerprint(
                    sr.get("rule_id", ""), sr.get("file_path", ""), sr.get("matched_text", "")
                )
            },
        ))

    return Sarif(
        version="2.1.0",
        schema_uri="https://json.schemastore.org/sarif-2.1.0.json",
        runs=[Run(
            tool=Tool(driver=ToolDriver(
                name="cloneguard",
                version=__version__,
                information_uri="https://github.com/prodnull/cloneguard",
                rules=rules,
            )),
            results=results,
        )],
    )

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| ~/.claude/settings.json -> integrity.py | Operator-controlled config file read for integrity verification |
| SARIF output -> GitHub Advanced Security | Serialized findings cross to external system; must not contain sensitive content |
| CLONEGUARD_SARIF_OUTPUT env var -> file write | Operator-controlled output path |
| pyproject.toml -> wheel build | Build config determines what ships in the package |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-03-01 | Tampering | integrity.py | mitigate | Check command pattern (`cloneguard hook-check --event`), not binary path. Warn but do not block -- user may have intentional config. Per D-13 and Pitfall 5. |
| T-03-02 | Information Disclosure | audit/sarif.py | mitigate | SARIF results contain file paths and matched pattern text but NOT full file content or raw tool input. Pattern match text is from regex match groups, not arbitrary content. |
| T-03-03 | Denial of Service | audit/sarif.py | mitigate | Cap results at 5,000 per GitHub limits. Log warning when truncated. Prevents unbounded memory consumption on large repos. |
| T-03-04 | Tampering | pyproject.toml | accept | Build config is developer-controlled. Package signing at publish time (future work) provides integrity at distribution. |
| T-03-05 | Spoofing | integrity.py | mitigate | Check that EACH expected event (InstructionsLoaded, PreToolUse, PostToolUse) has a CloneGuard hook entry. Missing events are reported as warnings, preventing silent removal of defense layers. |
</threat_model>
