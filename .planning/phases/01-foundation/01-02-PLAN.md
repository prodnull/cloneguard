```python
from __future__ import annotations
import datetime
from enum import Enum
from pydantic import BaseModel, ConfigDict, Field

class EventType(str, Enum):
    RISK_IDENTIFIED = "RISK_IDENTIFIED"
    SCAN_COMPLETE = "SCAN_COMPLETE"
    HOOK_INVOKED = "HOOK_INVOKED"

class SignalDetails(BaseModel):
    """Nested signal sub-object per D-05."""
    model_config = ConfigDict(frozen=True)
    pattern_matches: list[dict[str, str]] = Field(default_factory=list)
    pattern_severity: str = ""
    semantic_verdict: str = ""
    semantic_confidence: float = 0.0
    semantic_ood_distance: float = 0.0
    sequence_alerts: list[dict[str, str]] = Field(default_factory=list)
    summary: str = ""
    primary_rule_id: str = ""
    line_number: int = 0

class AuditEvent(BaseModel):
    """Structured audit event conforming to cloneguard/event/v1 schema (D-05)."""
    model_config = ConfigDict(frozen=True, populate_by_name=True)

    schema_version: str = "cloneguard/event/v1"       # D-07
    timestamp: datetime.datetime = Field(default_factory=lambda: datetime.datetime.now(datetime.timezone.utc))
    session_id: str
    agent_type: str = "claude-code"
    event_type: EventType
    tool_name: str
    tool_input_hash: str                                # SHA-256
    verdict: str                                        # "clean" | "suspicious" | "detected"
    confidence: float = 0.0
    signals: SignalDetails = Field(default_factory=SignalDetails)
    enforcement_action: str = "ALLOW"                   # Phase 1 default; Phase 2 adds CONSTRAIN/BLOCK
    constraints_applied: dict[str, list[str]] = Field(default_factory=dict)
    sandbox_adapter: str = "noop"
    outcome: str = "completed"
    policy_version: str = ""
    cloneguard_version: str
    source_path: str = ""

    def to_ndjson(self) -> str:
        """Serialize to a single NDJSON line (D-06)."""
        return self.model_dump_json(exclude_none=True) + "\n"
```

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| JSON stdin -> hooks.py shim | Untrusted agent-provided JSON from Claude Code hook protocol |
| hooks.py -> DetectionEngine | Shim passes parsed dict to engine; trust is in the engine's validation |
| DetectionEngine -> AuditEvent | Detection results cross into serialization layer; must not leak raw tool input |
| AuditEvent -> NDJSON file | Serialized events written to disk; file path from env var is operator-controlled |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-02-01 | Tampering | audit/ndjson.py | mitigate | NDJSON output NEVER writes to stdout (hook communication channel). Default to stderr or CLONEGUARD_NDJSON_OUTPUT file. |
| T-02-02 | Denial of Service | hooks.py _emit_audit_event | mitigate | Audit emission wrapped in try/except with pass -- audit failure never blocks agent operation. Lazy Pydantic import avoids cold-start penalty on critical path. |
| T-02-03 | Information Disclosure | audit/types.py | mitigate | AuditEvent stores `tool_input_hash` (SHA-256), not raw tool_input content. No sensitive content in audit events. |
| T-02-04 | Repudiation | audit/types.py | mitigate | Every AuditEvent includes timestamp, session_id, cloneguard_version for traceability. Schema version enables format evolution. |
| T-02-05 | Tampering | hooks.py shim | mitigate | Exit code mapping is hardcoded in shim, not configurable by repo content. DetectionResult.exit_code is set by engine, shim passes through directly. |
</threat_model>
