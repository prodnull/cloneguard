```python
class ScanMode(Enum):
    STRICT = "strict"
    STANDARD = "standard"
    LENIENT = "lenient"

class Verdict(Enum):
    CLEAN = "clean"
    DETECTED = "detected"
    SUSPICIOUS = "suspicious"

class PatternEngine:
    def scan(self, content: str, source_path: str, mode: ScanMode | None = None) -> ScanResult: ...
    # mode=None auto-detects from source_path:
    #   CLAUDE.md, .cursorrules, .claude/* -> STRICT
    #   test/, tests/ -> LENIENT
    #   everything else -> STANDARD

# YAML schema for rules:
# category: str
# description: str
# patterns:
#   - id: str (e.g. "LTL-001")
#     regex: str
#     severity: str (critical|high|medium)
#     description: str
#     false_positive_hint: str
#     modes: list[str]  # optional — omit to apply in all modes
```
